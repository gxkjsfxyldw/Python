```Python
from tkinter import *

window = Tk()

cv = Canvas(window,width=500,height=500)
cv.pack(fill=BOTH)
img = PhotoImage(file = r'E:\科师\2021春资料\课程\python\课件\第九章 图形用户界面设计\图\江南.gif')
cv.create_image((200,100),image = img)
a=Label(window,text='帐号')
a.pack()
b=Entry(window)
b.pack()
def pp():       

    print(b.get())
b1 = Button(window,text = '输出内容',command = pp)
b1.pack()
cv.create_window(50,350,window=b)
cv.create_window(50,50,window=a)
cv.create_window(120,50,height=None,width=100,window=b)
```
